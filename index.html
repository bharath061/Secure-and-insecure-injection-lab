      secure_agent.py
      

      SIDE_EFFECTS = {"send_email", "delete_files",
                "transfer_funds", "download_file"}

def run_agent(user_task, page_text):
    prompt = [
        {"role": "system", "content": SYSTEM},
        {"role": "user",   "content": user_task},        # trusted
        {"role": "tool",   "content": wrap_untrusted(page_text)},
    ]
    plan = llm(prompt)

    for call in plan.tool_calls:
        if call.name in SIDE_EFFECTS and call.name not in allowed_for(user_task):
            if not ask_user_in_chat(call):            # explicit yes
                log_blocked(call, source="page_text")
                continue
        TOOLS[call.name](**call.args)

    return plan.answer
      
      
      
    
      
  
      

  
      
    
      What this demo is not. The “model” here is a small rule-based stand-in, so you can see the structure without an API key. Real models are less predictable and real attackers are more creative. Scanning for suspicious text can be bypassed; the dependable layers are the labels on sources, the gate on side effects, and the explicit yes from you.
